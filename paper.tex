\begin{abstract}
A denominator records root multiplicities but forgets the Lie bracket.  We
first exhibit this loss in the smallest possible example.  We then fix an
even symmetrizable Borcherds--Cartan matrix and give a finite-height
recognition theorem for its positive Lie algebra.  The truncation takes place
in a symmetric monoidal category, so products and coproducts remain typed;
the theorem combines the Borcherds relations with root-space dimensions and
the Poincar\'e--Birkhoff--Witt theorem.  A comparison with a compact Hall
construction is stated separately as an open problem with explicit carriers
and maps.
\end{abstract}

\section{What a denominator forgets}

\begin{proposition}[Graded dimensions do not recover a bracket]
There are nonisomorphic positively graded Lie algebras with identical graded
dimensions.
\end{proposition}

\begin{proof}
Let $L$ have basis $x,y$ in degree one and $z$ in degree two, with
$[x,y]=z$ and $z$ central.  Let $L_0$ have the same graded vector space and
zero bracket.  Their graded dimensions agree, while
$[L,L]=\mathbb Cz$ and $[L_0,L_0]=0$.  Thus they are not isomorphic as Lie
algebras.
\end{proof}

Consequently, a product determined only by graded root multiplicities cannot
construct a Lie bracket.  Still less can it construct the correspondences,
orientation data, or convolution maps of a Hall algebra.

\section{The positive Borcherds presentation}

Let $\kk$ be a field of characteristic zero and let $I$ be finite.  An
\emph{even symmetrizable Borcherds--Cartan matrix} is an integral matrix
$A=(a_{ij})_{i,j\in I}$ such that
\begin{enumerate}
\item $a_{ii}=2$ or $a_{ii}\leq 0$;
\item $a_{ij}\leq 0$ for $i\neq j$, and $a_{ij}=0$ if and only if
      $a_{ji}=0$;
\item there are positive integers $d_i$ for which
      $d_i a_{ij}=d_j a_{ji}$.
\end{enumerate}
Here ``even'' means that every simple generator is assigned even parity; no
Lie-superalgebra signs occur.
Write $Q=\bigoplus_{i\in I}\mathbb Z\alpha_i$,
$Q_+=\bigoplus_{i\in I}\mathbb Z_{\geq0}\alpha_i$, and
$\htop(\sum_i n_i\alpha_i)=\sum_i n_i$.  Let $\mathfrak F$ be the free
$Q_+$-graded Lie algebra on generators $e_i$ of degree $\alpha_i$.  Let
$R_A\subset\mathfrak F$ be the homogeneous ideal generated by
\begin{align}
 (\operatorname{ad}e_i)^{1-a_{ij}}e_j&=0
 &&(i\neq j,\ a_{ii}=2),                                      \label{serre}\\
 [e_i,e_j]&=0
 &&(i\neq j,\ a_{ij}=0).                                      \label{orthogonal}
\end{align}
The positive Borcherds Lie algebra is
\[
 \mathfrak n_+(A)=\mathfrak F/R_A.
\]
These are the normalized positive defining relations corresponding to
Borcherds' original presentation~\cite{Borcherds1988}.  We use the integral
symmetrizable case, and finiteness of $I$ ensures that every
$Q_+$-homogeneous component is finite-dimensional.

\section{A truncation that preserves the operations}

For $N\geq1$, let $\mathsf{Gr}_{\leq N}(Q_+)$ be the category of
$Q_+$-graded $\kk$-vector spaces supported in heights at most $N$.  Its tensor
product is
\[
 (V\mathbin{\bar\otimes_N}W)_\beta
   =\bigoplus_{\gamma+\delta=\beta}V_\gamma\otimes W_\delta,
 \qquad \htop(\beta)\leq N.
\]
With the usual symmetry this is a symmetric monoidal category.  The functor
\[
 \tau_{\leq N}V=\bigoplus_{\htop(\beta)\leq N}V_\beta
\]
from nonnegatively $Q_+$-graded vector spaces to
$\mathsf{Gr}_{\leq N}(Q_+)$ is strong symmetric monoidal: a decomposition
$\beta=\gamma+\delta$ with $\htop(\beta)\leq N$ automatically has
$\htop(\gamma),\htop(\delta)\leq N$.  It therefore sends Lie algebras,
associative algebras, and bialgebras to objects of the same kind.  If
$M\leq N$, the evident functors satisfy
\[
 \tau_{\leq M}\tau_{\leq N}=\tau_{\leq M}.
\]
Thus $\tau_{\leq N}U(\mathfrak g)$ is a bialgebra object in the truncated
category even though the ordinary vector-space quotient obtained by
discarding degrees above $N$ need not be a coalgebra.

\begin{theorem}[Finite-height recognition]
Let $L=\bigoplus_{\beta\in Q_+\setminus\{0\}}L_\beta$ be a $Q_+$-graded Lie
algebra with finite-dimensional homogeneous pieces.  Suppose that elements
$x_i\in L_{\alpha_i}$ generate $L$ and satisfy \eqref{serre} and
\eqref{orthogonal}, with $e_i$ replaced by $x_i$.  Then there is a unique
surjective graded Lie homomorphism
\[
 \pi:\mathfrak n_+(A)\longrightarrow L,
 \qquad e_i\longmapsto x_i.
\]
If, for every $\beta$ with $\htop(\beta)\leq N$,
\begin{equation}
 \dim_\kk L_\beta=\dim_\kk\mathfrak n_+(A)_\beta,             \label{dimensions}
\end{equation}
then
\[
 \tau_{\leq N}\pi:\tau_{\leq N}\mathfrak n_+(A)
       \xrightarrow{\ \sim\ }\tau_{\leq N}L
\]
is an isomorphism of Lie algebra objects in
$\mathsf{Gr}_{\leq N}(Q_+)$.  Moreover
\begin{equation}
 \tau_{\leq N}U(\pi):\tau_{\leq N}U(\mathfrak n_+(A))
       \xrightarrow{\ \sim\ }\tau_{\leq N}U(L)             \label{pbwiso}
\end{equation}
is an isomorphism of bialgebra objects in that category.
\end{theorem}

\begin{proof}
The assignment $e_i\mapsto x_i$ gives a surjection
$\mathfrak F\twoheadrightarrow L$.  The displayed relations imply that its
kernel contains $R_A$, hence it factors through the asserted surjection
$\pi$.  For each $\beta$ of height at most $N$, the map $\pi_\beta$ is a
surjection between finite-dimensional spaces of the same dimension by
\eqref{dimensions}, and is therefore an isomorphism.  This proves the Lie
algebra statement.

Give each enveloping algebra its $Q_+$-grading.  By the
Poincar\'e--Birkhoff--Witt theorem, every homogeneous component of height at
most $N$ is built from homogeneous Lie components whose total height is at
most $N$.  On all of these components $\pi$ is an isomorphism, so $U(\pi)$ is
an isomorphism through height $N$.  Enveloping-algebra functoriality makes it
a bialgebra map, and strong monoidality of $\tau_{\leq N}$ gives
\eqref{pbwiso} in the stated category.
\end{proof}

\begin{remark}
Condition \eqref{dimensions} is not asked to recover the bracket.  The
relations first produce a surjective Lie map from an independently defined
Borcherds algebra; the dimensions then prove that this map has no additional
kernel through height $N$.
\end{remark}

\section{The compact Hall comparison}

Fix a proper Calabi--Yau category $\mathcal C$, a heart $\mathcal A$, and a
group homomorphism
\[
 \operatorname{cl}:K_0(\mathcal A)\longrightarrow Q
\]
whose value on every object of $\mathcal A$ lies in $Q_+$.  Let
$\mathfrak M_\beta$ be the moduli object of $E\in\mathcal A$ with
$\operatorname{cl}(E)=\beta$.  For $\beta,\gamma\in Q_+$, let
$\mathfrak E_{\beta,\gamma}$ parametrize short exact sequences, with
$E_\beta\in\mathfrak M_\beta$ and $E_\gamma\in\mathfrak M_\gamma$,
\[
 0\longrightarrow E_\gamma\longrightarrow E
  \longrightarrow E_\beta\longrightarrow0.
\]
Additivity in $K_0(\mathcal A)$ gives
$\operatorname{cl}(E)=\beta+\gamma$, and hence the extension correspondence
has the typed form
\begin{equation}
 \mathfrak M_\beta\times\mathfrak M_\gamma
 \xleftarrow{\ p_{\beta,\gamma}\ }
 \mathfrak E_{\beta,\gamma}
 \xrightarrow{\ q_{\beta,\gamma}\ }
 \mathfrak M_{\beta+\gamma}                                  \label{hallcorr}
\end{equation}
where $p_{\beta,\gamma}$ remembers $(E_\beta,E_\gamma)$ and
$q_{\beta,\gamma}$ remembers $E$.

Choose an orientation datum and a $\kk$-linear cohomology theory $H$, including
any componentwise cohomological shifts and Tate twists in this notation, and
put $H_\beta=H(\mathfrak M_\beta)$.  Assume that, for every
$\beta,\gamma\in Q_+$, the external-product morphism
\[
 \kappa_{\beta,\gamma}:
 H_\beta\otimes_\kk H_\gamma
 \xrightarrow{\ \sim\ }
 H(\mathfrak M_\beta\times\mathfrak M_\gamma)
\]
is an isomorphism and that the correspondence admits degree-zero maps
\[
 p_{\beta,\gamma}^*:H(\mathfrak M_\beta\times\mathfrak M_\gamma)
   \longrightarrow H(\mathfrak E_{\beta,\gamma}),
 \qquad
 (q_{\beta,\gamma})_!:H(\mathfrak E_{\beta,\gamma})
   \longrightarrow H_{\beta+\gamma}.
\]
Define
\begin{equation}
\begin{split}
 m_{\beta,\gamma}:H_\beta\otimes_\kk H_\gamma
 &\xrightarrow{\ \kappa_{\beta,\gamma}\ }
 H(\mathfrak M_\beta\times\mathfrak M_\gamma)
 \xrightarrow{\ p_{\beta,\gamma}^*\ } H(\mathfrak E_{\beta,\gamma})\\
 &\xrightarrow{\ (q_{\beta,\gamma})_!\ }H_{\beta+\gamma}.
                                                               \label{hallproduct}
\end{split}
\end{equation}
The associativity of these maps is a theorem only after the two iterated
extension correspondences and their base-change isomorphism have been
specified.  Cohomological Hall algebras constructed from such diagrams are
developed in~\cite{KontsevichSoibelman2011}.

\begin{problem}[Typed Hall--Borcherds comparison]
For a concrete compact Calabi--Yau category $\mathcal C$, make the choices in
\eqref{hallcorr} so that \eqref{hallproduct} defines a $Q_+$-graded
associative algebra $H$.  Exhibit even classes
$x_i\in H_{\alpha_i}$ and an even symmetrizable Borcherds--Cartan matrix
$A$.  For the Lie subalgebra $L\subset H$ generated by the $x_i$ under the
commutator, prove the relations \eqref{serre}--\eqref{orthogonal} and the
dimension equalities \eqref{dimensions}.  Require these statements for every
$N$, after applying $\tau_{\leq N}$, compatibly with all transition functors
$\tau_{\leq M}\tau_{\leq N}=\tau_{\leq M}$.  The theorem then identifies
each finite-height Lie algebra and enveloping bialgebra with the corresponding
positive Borcherds truncation.
\end{problem}
